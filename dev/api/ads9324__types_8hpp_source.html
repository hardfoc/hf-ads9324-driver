275[[nodiscard]] inline constexpr int32_t GainCorrectionFor(int32_t measured,
276 int32_t expected) noexcept {
277 if (measured == 0) {
278 return 0;
279 }
280 const int64_t num = static_cast<int64_t>(expected - measured) * 65536;
281 int64_t q = num / measured; // truncates toward zero
282 const int64_t r = num % measured;
283 if (2 * (r < 0 ? -r : r) >= (measured < 0 ? -static_cast<int64_t>(measured) : measured)) {
284 q += ((num < 0) != (measured < 0)) ? -1 : 1; // round half away from zero
285 }
286 return static_cast<int32_t>(q);
287}
