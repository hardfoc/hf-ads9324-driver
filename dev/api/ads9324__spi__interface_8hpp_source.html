45template <typename Port>
46inline bool Transfer(Port& port, const uint8_t* tx, uint8_t* rx, std::size_t len) {
47 using Result = TransferResult<Port>;
48 static_assert(std::is_void_v<Result> || kTransferReportsStatus<Port>,
49 "ads9324: the port's transfer(tx, rx, len) must return bool (true = frame "
50 "clocked, false = bus error), or void for a port that never reports errors");
51 if constexpr (std::is_void_v<Result>) {
52 port.transfer(tx, rx, len);
53 return true;
54 } else {
55 return static_cast<bool>(port.transfer(tx, rx, len));
