350[[nodiscard]] inline constexpr uint8_t FrameChannel(ChannelCountMode mode, uint8_t ch_sel,
351 uint8_t slot) noexcept {
352 const uint8_t n = FrameChannelCount(mode);
353 if (slot >= n) {
354 return kNumChannels;
355 }
356 if (n == kNumChannels) {
357 return slot;
358 }
359 const uint8_t half = static_cast<uint8_t>(n / 2u);
360 uint8_t group = 0;
361 switch (mode) {
362 case ChannelCountMode::Eight:
363 group = static_cast<uint8_t>((ch_sel >> 2) & 0x1u);
364 break;
365 case ChannelCountMode::Four:
366 group = static_cast<uint8_t>((ch_sel >> 1) & 0x3u);
367 break;
368 case ChannelCountMode::Two:
369 group = static_cast<uint8_t>(ch_sel & 0x7u);
370 break;
371 case ChannelCountMode::All16:
372 default:
373 break;
374 }
375 const uint8_t low = static_cast<uint8_t>(group * half);
376 if (slot < half) {
377 return static_cast<uint8_t>(low + slot);
378 }
379 return static_cast<uint8_t>(kNumChannels - low - half + (slot - half));
380}

388[[nodiscard]] inline constexpr uint16_t FrameChannelMask(ChannelCountMode mode,
389 uint8_t ch_sel) noexcept {
390 uint16_t mask = 0;
391 for (uint8_t slot = 0; slot < FrameChannelCount(mode); ++slot) {
392 mask = static_cast<uint16_t>(mask | (uint16_t{1} << FrameChannel(mode, ch_sel, slot)));
393 }
394 return mask;
395}
