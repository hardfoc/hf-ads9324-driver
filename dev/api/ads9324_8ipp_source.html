37template <typename SpiType, typename HostType>
38bool ADS9324<SpiType, HostType>::applyChannelDefaults() noexcept {
39 const ChannelConfig def = DefaultChannelConfig();
40 const uint16_t nibble = reg::EncodePgaNibble(def);
41 const auto pair = static_cast<uint16_t>((nibble << 8) | nibble);
42 uint16_t bw = 0;
43 for (uint8_t local = 0; local < 8; ++local) {
44 bw =
45 static_cast<uint16_t>(bw | ((static_cast<uint16_t>(def.bandwidth) & 0x3u) << (local * 2u)));
46 }
47 for (const RegisterBank bank : {RegisterBank::Ain1_8, RegisterBank::Ain9_16}) {
48 for (uint8_t local = 0; local < 8; local = static_cast<uint8_t>(local + 2u)) {
49 if (!writeRegister(bank, reg::PgaConfigAddr(local), pair)) {
50 return false;
51 }
52 }
53 if (!writeRegister(bank, reg::PGA_BW_SEL, bw)) {
54 return false;
55 }
56 const uint8_t first = (bank == RegisterBank::Ain1_8) ? 0u : 8u;
57 for (uint8_t ch = first; ch < first + 8u; ++ch) {
58 ch_cfg_[ch] = def;
59 }
60 }
61 return true;
62}

65template <typename SpiType, typename HostType>
66bool ADS9324<SpiType, HostType>::spiTransfer24(uint32_t tx24, uint32_t* rx24) noexcept {
67 const uint8_t tx[3] = {
68 static_cast<uint8_t>((tx24 >> 16) & 0xFFu),
69 static_cast<uint8_t>((tx24 >> 8) & 0xFFu),
70 static_cast<uint8_t>(tx24 & 0xFFu),
71 };
72 uint8_t rx[3] = {0, 0, 0};
73 const bool ok = detail::Transfer(spi_, tx, rx, reg::kSpiFrameBytes);
74 if (rx24 != nullptr) {
75 *rx24 = ok ? ((static_cast<uint32_t>(rx[0]) << 16) | (static_cast<uint32_t>(rx[1]) << 8) |
76 static_cast<uint32_t>(rx[2]))
77 : 0u;
78 }
79 return ok;
80}

112// Two-frame read: write GEN_CFG1 with REG_RD_ADD|REG_RD_EN, then clock a dummy
113// 24-bit frame. Payload is bits [23:8] of the second MISO word (Table 7-17).
114template <typename SpiType, typename HostType>
115bool ADS9324<SpiType, HostType>::readRegister(RegisterBank bank, uint8_t addr,
116 uint16_t& data) noexcept {
117 data = 0;
118 if (!selectBank(bank)) {
119 return false;
120 }
121 // Frame: write GEN_CFG1 with REG_RD_ADD and REG_RD_EN (Table 7-17).
122 if (!spiTransfer24(reg::PackWrite(reg::GEN_CFG1, reg::ReadCommand(addr)))) {
123 return false;
124 }
125 uint32_t rx = 0;
126 if (!spiTransfer24(0, &rx)) {
127 return false;
128 }
129 data = reg::UnpackRegData(rx);
130 return true;
131}

210template <typename SpiType, typename HostType>
211bool ADS9324<SpiType, HostType>::EnsureInitialized(bool force) noexcept {
212 if (initialized_ && !force) {
213 return true;
214 }
215 initialized_ = false;
216
217 host_.DelayMs(ADS9324_CFG::POWER_ON_DELAY_MS);
218
219 if (!resetChip()) {
220 ++errors_;
221 return false;
222 }
223
224 if (ADS9324_CFG::REQUIRE_DEVICE_ID) {

226 if (id != kDeviceIdAds9324) {
227 ++errors_;
228 return false;

234 ++errors_;

255template <typename SpiType, typename HostType>
256bool ADS9324<SpiType, HostType>::ConfigureChannel(uint8_t channel,
257 const ChannelConfig& cfg) noexcept {
258 if (channel >= kNumChannels) {
259 return false;
260 }
261 const uint8_t local = reg::LocalChannel(channel);
262 const RegisterBank bank = reg::BankForChannel(channel);
263 const uint8_t addr = reg::PgaConfigAddr(local);
264
265 uint16_t current = 0;
266 if (!readRegister(bank, addr, current)) {
267 return false;
268 }
269
270 const uint16_t nibble = reg::EncodePgaNibble(cfg);
271 if ((local & 1u) == 0) {
272 current = static_cast<uint16_t>((current & 0xFF00u) | (nibble & 0x00FFu));
273 } else {
274 current = static_cast<uint16_t>((current & 0x00FFu) | (static_cast<uint16_t>(nibble) << 8));
275 }
276 if (!writeRegister(bank, addr, current)) {
277 return false;
278 }
279
280 uint16_t bw = 0;
281 if (!readRegister(bank, reg::PGA_BW_SEL, bw)) {
282 return false;
283 }
284 const uint8_t shift = static_cast<uint8_t>(local * 2u);
285 bw = static_cast<uint16_t>((bw & static_cast<uint16_t>(~(0x3u << shift))) |
286 (static_cast<uint16_t>(static_cast<uint16_t>(cfg.bandwidth) & 0x3u)
287 << shift));
288 if (!writeRegister(bank, reg::PGA_BW_SEL, bw)) {
289 return false;
290 }
291
292 ch_cfg_[channel] = cfg;
293 return true;
294}

390template <typename SpiType, typename HostType>
391bool ADS9324<SpiType, HostType>::SetOversampling(OversamplingRatio osr,
392 OversamplingClock clock) noexcept {
393 if (static_cast<uint8_t>(osr) > static_cast<uint8_t>(OversamplingRatio::X128)) {
394 return false;

396 const bool enable = osr != OversamplingRatio::None;
397 // §7.3.11.2: initialize GEN_CFG5 before using the digital filters, OVR_EN first.
398 if (enable && !avg_mode_ready_) {

445template <typename SpiType, typename HostType>
446bool ADS9324<SpiType, HostType>::RunAdcCalibration(AdcCalMode mode, InputType ain1_8,
447 InputType ain9_16) noexcept {
448 if (!initialized_ ||
449 static_cast<uint8_t>(mode) > static_cast<uint8_t>(AdcCalMode::OffsetAndGain)) {
450 return false;
451 }
452 const uint16_t cal =
453 reg::AdcCalValue(mode, ain1_8 == InputType::Differential, ain9_16 == InputType::Differential);
454 if (mode == AdcCalMode::Disabled) {
455 if (!writeRegister(RegisterBank::Common, reg::ADC_CAL, cal)) {
456 return false;
457 }
458 adc_cal_mode_ = mode;
459 return true;
460 }
461 // Step 1 (and "disable digital filters" before ADC_CAL): only INT_TRIG_MODE set.
462 if (!writeDigFilter(reg::DIG_FILTER_INT_TRIG_MODE, false) ||
463 !writeRegister(RegisterBank::Common, reg::ADC_CAL, cal) ||
464 !writeRegister(RegisterBank::Common, reg::ADC_CAL,
465 static_cast<uint16_t>(cal | reg::ADC_CAL_TRIG))) {
466 return false;
467 }
468 host_.DelayUs(10);
469 if (!writeRegister(RegisterBank::Common, reg::ADC_CAL, cal)) {
470 return false;
471 }
472 host_.ConvstWrite(true);
473 host_.DelayUs(1);
474 host_.ConvstWrite(false);
475 host_.DelayUs(1);
476 host_.ConvstWrite(true);
477
478 host_.DelayMs(ADS9324_CFG::ADC_CAL_WAIT_MS);
479 bool done = false;
480 bool bus_ok = true;
481 for (uint32_t ms = 0; ms <= ADS9324_CFG::ADC_CAL_TIMEOUT_MS; ++ms) {
482 uint16_t status = 0;
483 if (!readRegister(RegisterBank::Common, reg::DEVICE_STATUS, status)) {
484 bus_ok = false;
485 break;
486 }
487 if ((status & reg::DEVICE_STATUS_CALIB_BUSY) == 0) {
488 done = true;
489 break;
490 }
491 host_.DelayMs(1);
492 }
493 host_.ConvstWrite(false);
494 // Put the oversampling filter back (SYSREF pulse resets its accumulator).
495 const bool restored = bus_ok && writeDigFilter(reg::DigFilterValue(osr_, osr_clock_),
496 osr_ != OversamplingRatio::None);
497 if (!done || !restored) {
498 ++errors_;
499 return false;
500 }
501 adc_cal_mode_ = mode;
502 return true;
503}

520template <typename SpiType, typename HostType>
521bool ADS9324<SpiType, HostType>::ProgramWindowComparator(uint8_t channel, int8_t high_th,
522 int8_t low_th, bool enable) noexcept {
523 if (channel >= kNumChannels) {
524 return false;
525 }
526 const RegisterBank bank = reg::BankForChannel(channel);
527 const uint8_t local = reg::LocalChannel(channel);
528 const uint16_t th = (static_cast<uint16_t>(static_cast<uint8_t>(high_th)) << 8) |
529 static_cast<uint16_t>(static_cast<uint8_t>(low_th));
530 if (!writeRegister(bank, reg::DwcThAddr(local), th)) {
531 return false;
532 }
533 uint16_t cfg = 0;
534 if (!readRegister(bank, reg::DWC_CFG, cfg)) {
535 return false;
536 }
537 const uint16_t bit = static_cast<uint16_t>(1u << local);
538 if (enable) {
539 cfg = static_cast<uint16_t>(cfg | bit);
540 } else {
541 cfg = static_cast<uint16_t>(cfg & static_cast<uint16_t>(~bit));
542 }
543 return writeRegister(bank, reg::DWC_CFG, cfg);
544}

557template <typename SpiType, typename HostType>
558bool ADS9324<SpiType, HostType>::pulseConvstAndWait() noexcept {
559 const uint8_t samples = OversamplingSamples(osr_);
560 const bool external = osr_clock_ == OversamplingClock::External;
561 const uint8_t edges = external ? samples : uint8_t{1};
562 for (uint8_t i = 0; i < edges; ++i) {
563 if (i != 0) {
564 host_.DelayUs(1);
565 }
566 host_.ConvstWrite(true);
567 host_.DelayUs(1);
568 host_.ConvstWrite(false);
569 }
570
571 const uint32_t scale = external ? 1u : samples;
572 if (host_.HasDrdy()) {
573 const uint32_t budget = ADS9324_CFG::DRDY_TIMEOUT_US * scale;
574 for (uint32_t i = 0; i < budget; ++i) {
575 if (host_.DrdyRead()) {
576 return true;
577 }
578 host_.DelayUs(1);
579 }
580 return false;
581 }
582 host_.DelayUs(ADS9324_CFG::CONV_FALLBACK_US * scale);
583 return true;
584}

592template <typename SpiType, typename HostType>
593Snapshot ADS9324<SpiType, HostType>::ReadSnapshot() noexcept {
594 Snapshot snap{};
595 if (!initialized_) {
596 snap.error = Error::NotInitialized;
597 ++errors_;
598 return snap;
599 }
600 if (!pulseConvstAndWait()) {
601 snap.error = Error::Timeout;
602 ++errors_;
603 return snap;
604 }
605
606 const uint8_t n = activeChannelCount();
607 const std::size_t bytes_per = (length_ == DoutLength::Bits24) ? 3u : 2u;
608 uint8_t rx[kNumChannels * 3]{};
609 const uint8_t tx[kNumChannels * 3]{};
610 if (!detail::Transfer(spi_, tx, rx, static_cast<std::size_t>(n) * bytes_per)) {
611 snap.error = Error::SpiError;
612 ++errors_;
613 return snap;
614 }
615
616 for (uint8_t i = 0; i < n && i < kNumChannels; ++i) {
617 const std::size_t at = static_cast<std::size_t>(i) * bytes_per;
618 const auto raw = static_cast<uint16_t>((static_cast<uint16_t>(rx[at]) << 8) | rx[at + 1]);
619 const uint8_t ch = reg::FrameChannel(ch_count_mode_, ch_sel_, i);
620 if (ch >= kNumChannels) { // FrameChannel's "no channel" value; cannot happen for i < n
621 continue;
622 }
623 const int16_t twos = WireWordToTwos(raw, format_);
624 snap.count[ch] = twos;
625 snap.voltage[ch] = CountToVoltage(ch, twos);
626 snap.valid_mask = static_cast<uint16_t>(snap.valid_mask | (uint16_t{1} << ch));
627 }
628 ++snapshots_;
629 return snap;
630}

632template <typename SpiType, typename HostType>
633ReadResult ADS9324<SpiType, HostType>::ReadChannel(uint8_t channel) noexcept {
634 ReadResult r{};
635 r.channel = channel;
636 if (channel >= kNumChannels) {
637 r.error = Error::InvalidChannel;
638 ++errors_;
639 return r;
640 }
641 const Snapshot snap = ReadSnapshot();
642 r.error = snap.error;
643 if (!snap.ok() || !snap.hasChannel(channel)) {
644 if (r.error == Error::Ok) {
645 r.error = Error::SpiError;
646 }
647 return r;
648 }
649 r.count = snap.count[channel];
650 r.raw = static_cast<uint16_t>(snap.count[channel]);
651 r.voltage = snap.voltage[channel];
652 return r;
653}
