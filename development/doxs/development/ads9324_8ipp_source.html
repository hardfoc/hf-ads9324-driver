35template <typename SpiType, typename HostType>
36uint32_t ADS9324<SpiType, HostType>::spiTransfer24(uint32_t tx24) noexcept {
37 uint8_t tx[3] = {
38 static_cast<uint8_t>((tx24 >> 16) & 0xFFu),
39 static_cast<uint8_t>((tx24 >> 8) & 0xFFu),
40 static_cast<uint8_t>(tx24 & 0xFFu),
41 };
42 uint8_t rx[3] = {0, 0, 0};
43 spi_.transfer(tx, rx, 3);
44 return (static_cast<uint32_t>(rx[0]) << 16) | (static_cast<uint32_t>(rx[1]) << 8) |
45 static_cast<uint32_t>(rx[2]);
46}

73// Two-frame read: write GEN_CFG1 with REG_RD_ADD|REG_RD_EN, then clock a dummy
74// 24-bit frame. Payload is bits [23:8] of the second MISO word (Table 7-17).
75template <typename SpiType, typename HostType>
76bool ADS9324<SpiType, HostType>::readRegister(RegisterBank bank, uint8_t addr,
77 uint16_t& data) noexcept {
78 if (!selectBank(bank)) {
79 return false;
80 }
81 // Frame: write GEN_CFG1 with REG_RD_ADD and REG_RD_EN (Table 7-17).
82 spiTransfer24(reg::PackWrite(reg::GEN_CFG1, reg::ReadCommand(addr)));
83 const uint32_t rx = spiTransfer24(0);
84 data = reg::UnpackRegData(rx);
85 return true;
86}

153template <typename SpiType, typename HostType>
154bool ADS9324<SpiType, HostType>::ConfigureChannel(uint8_t channel,
155 const ChannelConfig& cfg) noexcept {
156 if (channel >= kNumChannels) {
157 return false;
158 }
159 const uint8_t local = reg::LocalChannel(channel);
160 const RegisterBank bank = reg::BankForChannel(channel);
161 const uint8_t addr = reg::PgaConfigAddr(local);
162
163 uint16_t current = 0;
164 if (!readRegister(bank, addr, current)) {
165 return false;
166 }
167
168 const uint16_t nibble = reg::EncodePgaNibble(cfg);
169 if ((local & 1u) == 0) {
170 current = static_cast<uint16_t>((current & 0xFF00u) | (nibble & 0x00FFu));
171 } else {
172 current = static_cast<uint16_t>((current & 0x00FFu) | (static_cast<uint16_t>(nibble) << 8));
173 }
174 if (!writeRegister(bank, addr, current)) {
175 return false;
176 }
177
178 uint16_t bw = 0;
179 if (!readRegister(bank, reg::PGA_BW_SEL, bw)) {
180 return false;
181 }
182 const uint8_t shift = static_cast<uint8_t>(local * 2u);
183 bw = static_cast<uint16_t>((bw & static_cast<uint16_t>(~(0x3u << shift))) |
184 (static_cast<uint16_t>(static_cast<uint16_t>(cfg.bandwidth) & 0x3u)
185 << shift));
186 if (!writeRegister(bank, reg::PGA_BW_SEL, bw)) {
187 return false;
188 }
189
190 ch_cfg_[channel] = cfg;
191 return true;
192}

270template <typename SpiType, typename HostType>
271bool ADS9324<SpiType, HostType>::ProgramWindowComparator(uint8_t channel, int8_t high_th,
272 int8_t low_th, bool enable) noexcept {
273 if (channel >= kNumChannels) {
274 return false;
275 }
276 const RegisterBank bank = reg::BankForChannel(channel);
277 const uint8_t local = reg::LocalChannel(channel);
278 const uint16_t th = (static_cast<uint16_t>(static_cast<uint8_t>(high_th)) << 8) |
279 static_cast<uint16_t>(static_cast<uint8_t>(low_th));
280 if (!writeRegister(bank, reg::DwcThAddr(local), th)) {
281 return false;
282 }
283 uint16_t cfg = 0;
284 if (!readRegister(bank, reg::DWC_CFG, cfg)) {
285 return false;
286 }
287 const uint16_t bit = static_cast<uint16_t>(1u << local);
288 if (enable) {
289 cfg = static_cast<uint16_t>(cfg | bit);
290 } else {
291 cfg = static_cast<uint16_t>(cfg & static_cast<uint16_t>(~bit));
292 }
293 return writeRegister(bank, reg::DWC_CFG, cfg);
294}

311template <typename SpiType, typename HostType>
312bool ADS9324<SpiType, HostType>::pulseConvstAndWait() noexcept {
313 host_.ConvstWrite(true);
314 host_.DelayUs(1);
315 host_.ConvstWrite(false);
316
317 if (host_.HasDrdy()) {
318 const uint32_t budget = ADS9324_CFG::DRDY_TIMEOUT_US;
319 for (uint32_t i = 0; i < budget; ++i) {
320 if (host_.DrdyRead()) {
321 return true;
322 }
323 host_.DelayUs(1);
324 }
325 return false;
326 }
327 host_.DelayUs(ADS9324_CFG::CONV_FALLBACK_US);
328 return true;
329}

336template <typename SpiType, typename HostType>
337Snapshot ADS9324<SpiType, HostType>::ReadSnapshot() noexcept {
338 Snapshot snap{};
339 if (!initialized_) {
340 snap.error = Error::NotInitialized;
341 ++errors_;
342 return snap;
343 }
344 if (!pulseConvstAndWait()) {
345 snap.error = Error::Timeout;
346 ++errors_;
347 return snap;
348 }
349
350 const uint8_t n = activeChannelCount();
351 const std::size_t bytes_per = (length_ == DoutLength::Bits24) ? 3u : 2u;
352 uint8_t rx[kNumChannels * 3]{};
353 uint8_t tx[kNumChannels * 3]{};
354 spi_.transfer(tx, rx, static_cast<std::size_t>(n) * bytes_per);
355
356 for (uint8_t i = 0; i < n && i < kNumChannels; ++i) {
357 uint16_t raw = 0;
358 if (bytes_per == 2) {
359 raw = static_cast<uint16_t>((static_cast<uint16_t>(rx[i * 2]) << 8) | rx[i * 2 + 1]);
360 } else {
361 raw = static_cast<uint16_t>((static_cast<uint16_t>(rx[i * 3]) << 8) | rx[i * 3 + 1]);
362 }
363 const int16_t twos = WireWordToTwos(raw, format_);
364 snap.count[i] = twos;
365 snap.voltage[i] = CountToVoltage(i, twos);
366 snap.valid_mask = static_cast<uint16_t>(snap.valid_mask | (uint16_t{1} << i));
367 }
368 ++snapshots_;
369 return snap;
370}

372template <typename SpiType, typename HostType>
373ReadResult ADS9324<SpiType, HostType>::ReadChannel(uint8_t channel) noexcept {
374 ReadResult r{};
375 r.channel = channel;
376 if (channel >= kNumChannels) {
377 r.error = Error::InvalidChannel;
378 ++errors_;
379 return r;
380 }
381 const Snapshot snap = ReadSnapshot();
382 r.error = snap.error;
383 if (!snap.ok() || !snap.hasChannel(channel)) {
384 if (r.error == Error::Ok) {
385 r.error = Error::SpiError;
386 }
387 return r;
388 }
389 r.count = snap.count[channel];
390 r.raw = static_cast<uint16_t>(snap.count[channel]);
391 r.voltage = snap.voltage[channel];
392 return r;
393}
