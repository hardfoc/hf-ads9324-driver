194[[nodiscard]] inline constexpr uint16_t EncodePgaNibble(const ChannelConfig& cfg) noexcept {
195 const uint16_t cme = cfg.common_mode_correct ? 1u : 0u;
196 const uint16_t cm = static_cast<uint16_t>(cfg.type) & 0x7u;
197 const uint16_t ir = static_cast<uint16_t>(cfg.range) & 0x7u;
198 return static_cast<uint16_t>((cme << 7) | (cm << 4) | ir);
199}

209[[nodiscard]] inline constexpr uint16_t GenCfg3Value(DoutLaneMode lanes,
210 DoutLength length,
211 bool data_on_sdout,
212 DataFormat format) noexcept {
213 uint16_t v = 0;
214 if (format == DataFormat::OffsetBinary) {
215 v |= GEN_CFG3_EN_OFS_BINARY;
216 }
217 v |= static_cast<uint16_t>((static_cast<uint16_t>(lanes) & 0x3u) << 4);
218 v |= static_cast<uint16_t>((static_cast<uint16_t>(length) & 0x3u) << 2);
219 if (data_on_sdout) {
220 v |= GEN_CFG3_ADC_DATA_SDOUT_EN;
221 }
222 return v;
223}
